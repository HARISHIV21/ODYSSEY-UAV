import asyncio

from MissionComputer.flight_manager import FlightManager
from Navigation.mission_loader import MissionLoader


async def main():

    loader = MissionLoader("Data/mission.json")
    mission = loader.load()

    fm = FlightManager()

    await fm.connect()
    await fm.wait_until_ready()

    await fm.arm()

    await fm.takeoff(
        mission["takeoff_altitude"]
    )

    telemetry_task = asyncio.create_task(
        fm.telemetry.monitor()
    )

    print("\n==============================")
    print("MISSION STARTED")
    print("==============================\n")

    # Allow takeoff to stabilize before Offboard starts.
    await asyncio.sleep(10)

    await fm.offboard.start(
        mission["takeoff_altitude"]
    )

    total = len(mission["waypoints"])

    for index, wp in enumerate(
        mission["waypoints"],
        start=1
    ):

        print(f"\n[MISSION] Waypoint {index}/{total}")

        await fm.offboard.goto(
            wp["north"],
            wp["east"],
            wp["down"],
            wp["yaw"]
        )

    await fm.offboard.stop()

    await fm.land()

    telemetry_task.cancel()

    print("\n==============================")
    print("MISSION COMPLETE")
    print("==============================\n")


if __name__ == "__main__":
    asyncio.run(main())

import asyncio
from mavsdk.offboard import PositionNedYaw, OffboardError


class OffboardController:

    def __init__(self, drone):
        self.drone = drone

    async def start(self, altitude=10.0):

        print("[OFFBOARD] Initializing...")

        # Send an initial setpoint at the desired altitude.
        # In NED coordinates, Down is negative above the home position.
        await self.drone.offboard.set_position_ned(
            PositionNedYaw(
                north_m=0.0,
                east_m=0.0,
                down_m=-altitude,
                yaw_deg=0.0
            )
        )

        await asyncio.sleep(0.2)

        try:
            await self.drone.offboard.start()
            print("[OFFBOARD] Started")

        except OffboardError as error:
            print(f"[OFFBOARD] Failed: {error}")
            raise

    async def goto(
        self,
        north,
        east,
        down,
        yaw=0.0,
        hold_time=5
    ):

        print(
            f"[OFFBOARD] Going to "
            f"N={north:.1f} "
            f"E={east:.1f} "
            f"D={down:.1f}"
        )

        await self.drone.offboard.set_position_ned(
            PositionNedYaw(
                north_m=north,
                east_m=east,
                down_m=down,
                yaw_deg=yaw
            )
        )

        await asyncio.sleep(hold_time)

    async def stop(self):

        print("[OFFBOARD] Stopping")

        try:
            await self.drone.offboard.stop()

        except OffboardError:
            pass

from enum import Enum


class UAVState(Enum):
    IDLE = 0
    INITIALIZING = 1
    CONNECTED = 2
    ARMED = 3
    TAKEOFF = 4
    MISSION = 5
    RTL = 6
    LANDING = 7
    COMPLETE = 8


class MissionCore:

    def __init__(self):
        self.state = UAVState.IDLE

    def change_state(self, state):
        print(f"[CORE] {self.state.name} -> {state.name}")
        self.state = state

    def initialize(self):
        self.change_state(UAVState.INITIALIZING)

    def connect(self):
        self.change_state(UAVState.CONNECTED)

    def arm(self):
        self.change_state(UAVState.ARMED)

    def takeoff(self):

    def mission(self):

    def rtl(self):
        self.change_state(UAVState.RTL)

    def land(self):
        self.change_state(UAVState.LANDING)

    def finish(self):

from datetime import datetime
from pathlib import Path


class MissionLogger:

    def __init__(self):
        Path("../Logs").mkdir(exist_ok=True)

        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")

        self.file = open(
            f"../Logs/mission_{timestamp}.log",
            "w"
        )

    def log(self, message):

        line = (
            f"[{datetime.now().strftime('%H:%M:%S')}] "
            f"{message}"
        )

        print(line)

        self.file.write(line + "\n")
        self.file.flush()

    def close(self):
        self.file.close()

import asyncio
from mavsdk import System

from MissionComputer.offboard_controller import OffboardController
from MissionComputer.telemetry_manager import TelemetryManager


class FlightManager:

    def __init__(self):
        self.drone = System()
        self.offboard = OffboardController(self.drone)
        self.telemetry = TelemetryManager(self.drone)

    async def connect(self):

        print("[MISSION] Connecting to PX4...")

        await self.drone.connect(
            system_address="udpin://0.0.0.0:14540"
        )

        async for state in self.drone.core.connection_state():
            if state.is_connected:
                print("[MISSION] Connected")
                break

    async def wait_until_ready(self):

        print("[MISSION] Waiting for vehicle health...")

        async for health in self.drone.telemetry.health():
            if (
                health.is_global_position_ok
                and health.is_home_position_ok
            ):
                print("[MISSION] Vehicle Ready")
                break

    async def arm(self):
        print("[MISSION] Arming...")
        await self.drone.action.arm()
        print("[MISSION] Armed")

    async def takeoff(self, altitude=10.0):
        print(f"[MISSION] Setting takeoff altitude to {altitude} m...")
        await self.drone.action.set_takeoff_altitude(altitude)
        print("[MISSION] Taking off...")
        await self.drone.action.takeoff()
    async def wait_until_altitude(self, target_altitude):

        print(f"[MISSION] Waiting to reach {target_altitude:.1f} m...")

        async for position in self.drone.telemetry.position():

            altitude = position.relative_altitude_m

            print(
                f"[MISSION] Current Altitude: {altitude:.2f} m",
                end="\r"
            )

            if altitude >= target_altitude - 0.5:

                print()

                print("[MISSION] Target altitude reached.")

                break

    async def land(self):
        print("[MISSION] Landing...")
        await self.drone.action.land()

    async def rtl(self):
        print("[MISSION] Returning to Launch...")
        await self.drone.action.return_to_launch()

    async def hold(self):
        print("[MISSION] Hold Position...")
        await self.drone.action.hold()

sys.path.append("../Navigation")

from mission_planner import MissionPlanner
from controller import FlightController
from battery import Battery


class MissionExecutor:

    def __init__(self):

        self.controller = FlightController()
        self.planner = MissionPlanner()
        self.battery = Battery()

    def setup(self):

        self.planner.add_waypoint(10, 5, 10)
        self.planner.add_waypoint(20, 10, 15)
        self.planner.add_waypoint(30, 15, 20)

    def execute(self):

        self.controller.arm()
        self.controller.takeoff()

        for waypoint in self.planner.waypoints:

            self.battery.consume()
            self.battery.check()

            print(f"[MISSION] Flying to {waypoint}")

            self.controller.move_to(*waypoint)

        self.controller.land()


if __name__ == "__main__":

    mission = MissionExecutor()

    mission.setup()

    mission.execute()

class Battery:

    def __init__(self):
        self.level = 100

    def consume(self):
        self.level -= 5

    def check(self):
        print(f"[BATTERY] {self.level}%")

        if self.level <= 20:
            print("[BATTERY] WARNING - LOW BATTERY")


if __name__ == "__main__":

    battery = Battery()

    for _ in range(18):
        battery.consume()
        battery.check()

import random

class IMU:

    def read(self):
        return {
            "roll": round(random.uniform(-5,5),2),
            "pitch": round(random.uniform(-5,5),2),
            "yaw": round(random.uniform(0,360),2)
        }


if __name__ == "__main__":

    imu = IMU()

    print(imu.read())
