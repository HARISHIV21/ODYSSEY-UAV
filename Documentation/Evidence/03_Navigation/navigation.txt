import asyncio
import math

from Navigation.distance import DistanceCalculator


class NavigationEngine:

    def __init__(self, drone, tolerance=1.0):
        self.drone = drone
        self.tolerance = tolerance
        self.distance = DistanceCalculator()

    async def wait_until_arrived(self, target, timeout=60.0):

        if (
            isinstance(timeout, bool)
            or not isinstance(timeout, (int, float))
            or not math.isfinite(timeout)
            or timeout <= 0
        ):
            raise ValueError("Navigation timeout must be a positive finite number")

        print(
            f"[NAV] Tracking waypoint "
            f"N:{target['north']} "
            f"E:{target['east']} "
            f"D:{target['down']}"
        )

        try:
            async with asyncio.timeout(timeout):
                async for data in self.drone.telemetry.position_velocity_ned():

                    current = data.position

                    distance_3d = self.distance.distance_3d(
                        current, target
                    )

                    horizontal = self.distance.horizontal(
                        current, target
                    )

                    vertical = self.distance.vertical(
                        current, target
                    )

                    print(
                        f"[NAV] Distance: {distance_3d:.2f} m | "
                        f"Horizontal: {horizontal:.2f} m | "
                        f"Vertical: {vertical:.2f} m"
                    )

                    if distance_3d <= self.tolerance:
                        print(
                            f"[NAV] Waypoint reached "
                            f"({distance_3d:.2f} m)"
                        )
                        return True

                    await asyncio.sleep(0.2)

        except TimeoutError as exc:
            raise TimeoutError(
                f"Waypoint arrival not confirmed within {timeout} seconds"
            ) from exc

        raise RuntimeError(
            "Position telemetry stream ended before waypoint arrival"
        )

class WaypointTracker:

    def __init__(self, mission):

        self.waypoints = mission["waypoints"]
        self.current = 0

    def total(self):

        return len(self.waypoints)

    def has_next(self):

        return self.current < self.total()

    def current_waypoint(self):

        if not self.has_next():
            return None

        return self.waypoints[self.current]

    def advance(self):

        if self.has_next():
            self.current += 1

    def completed(self):

        return self.current

    def remaining(self):

        return self.total() - self.current

    def progress(self):

        if self.total() == 0:
            return 100.0

        return (
            self.completed()
            / self.total()
        ) * 100.0

    def print_progress(self):

        print("\n========== MISSION PROGRESS ==========")

        print(
            f"Completed : {self.completed()}/{self.total()}"
        )

        print(
            f"Remaining : {self.remaining()}"
        )

        print(
            f"Progress  : {self.progress():.1f}%"
        )

        print("======================================")

import math


class DistanceCalculator:

    @staticmethod
    def distance_3d(current, target):

        dx = current.north_m - target["north"]
        dy = current.east_m - target["east"]
        dz = current.down_m - target["down"]

        return math.sqrt(dx * dx + dy * dy + dz * dz)

    @staticmethod
    def horizontal(current, target):

        dx = current.north_m - target["north"]
        dy = current.east_m - target["east"]

        return math.sqrt(dx * dx + dy * dy)

    @staticmethod
    def vertical(current, target):

        return abs(current.down_m - target["down"])
