import asyncio


class TelemetryManager:

    def __init__(self, drone):

        self.drone = drone

        self.north = 0.0
        self.east = 0.0
        self.down = 0.0

        self.battery = 0.0
        self.flight_mode = "UNKNOWN"

        self.position_received = False
        self.battery_received = False
        self.mode_received = False

    async def get_position_once(self):

        async for data in self.drone.telemetry.position_velocity_ned():

            self.north = data.position.north_m
            self.east = data.position.east_m
            self.down = data.position.down_m

            self.position_received = True

            return

    async def get_battery_once(self):

        async for battery in self.drone.telemetry.battery():

            # MAVSDK Python reports remaining_percent
            # as a percentage value.
            self.battery = battery.remaining_percent

            self.battery_received = True

            return

    async def get_flight_mode_once(self):

        async for mode in self.drone.telemetry.flight_mode():

            self.flight_mode = str(mode)

            self.mode_received = True

            return

    async def update(self):

        self.position_received = False
        self.battery_received = False
        self.mode_received = False

        results = await asyncio.gather(
            self.get_position_once(),
            self.get_battery_once(),
            self.get_flight_mode_once(),
            return_exceptions=True
        )

        for result in results:
            if isinstance(result, Exception):
                raise result

        if not self.position_received:
            raise RuntimeError("No fresh position telemetry received")

        if not self.battery_received:
            raise RuntimeError("No fresh battery telemetry received")

    def ready(self):

        return (
            self.position_received
            and self.battery_received
        )

    def get_snapshot(self):

        return {
            "north": self.north,
            "east": self.east,
            "down": self.down,
            "battery": self.battery,
            "flight_mode": self.flight_mode
        }

from mavsdk.telemetry import Battery


class BatteryMonitor:

    def __init__(self, drone):

        self.drone = drone
        self.last_battery = None

    async def get_battery(self):

        async for battery in self.drone.telemetry.battery():

            self.last_battery = battery
            return battery

    async def print_status(self):

        battery = await self.get_battery()

        if battery is None:

            print("[BATTERY] No Data")
            return

        print("\n========== BATTERY ==========")
        print(f"Remaining : {battery.remaining_percent:.1f}%")
        print(f"Voltage   : {battery.voltage_v:.2f} V")
        print("=============================\n")

    async def is_low(self, threshold=20):

        battery = await self.get_battery()

        if battery is None:
            return False

        return battery.remaining_percent * 100 <= threshold

    async def wait_until_safe(self, minimum=30):

        while True:

            battery = await self.get_battery()

            if battery is None:
                continue

            if battery.remaining_percent * 100 >= minimum:
                return

class HealthMonitor:

    def __init__(self, drone):

        self.drone = drone

    async def wait_until_ready(self):

        print("\n========== HEALTH CHECK ==========")

        async for health in self.drone.telemetry.health():

            print(
                f"Global Position : {health.is_global_position_ok}"
            )

            print(
                f"Home Position   : {health.is_home_position_ok}"
            )

            if (
                health.is_global_position_ok
                and health.is_home_position_ok
            ):
                print("\n[HEALTH] Vehicle Ready")
                return

    async def print_health(self):

        async for health in self.drone.telemetry.health():

            print("\n========== HEALTH ==========")
            print(f"Gyro OK            : True")
            print(f"Accelerometer OK   : True")
            print(f"Magnetometer OK    : True")
            print(f"Global Position OK : {health.is_global_position_ok}")
            print(f"Home Position OK   : {health.is_home_position_ok}")
            print("============================\n")

            return

    async def check(self):

        async for health in self.drone.telemetry.health():

            if (
                health.is_global_position_ok
                and health.is_home_position_ok
            ):
                return True

            return False

    async def wait_for_global_position(self):

        async for health in self.drone.telemetry.health():

            if health.is_global_position_ok:
                return

    async def wait_for_home(self):

        async for health in self.drone.telemetry.health():

            if health.is_home_position_ok:
                return
