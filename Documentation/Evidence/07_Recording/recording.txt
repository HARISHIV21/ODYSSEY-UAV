import csv
import os
from datetime import datetime


class FlightRecorder:

    def __init__(self):

        os.makedirs("Logs", exist_ok=True)

        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")

        self.filename = f"Logs/flight_data_{timestamp}.csv"

        self.file = open(self.filename, "w", newline="")

        self.writer = csv.writer(self.file)

        self.writer.writerow([
            "Time",
            "North",
            "East",
            "Down",
            "Battery(%)",
            "FlightMode"
        ])

        self.closed = False

        print(f"[RECORDER] Logging to {self.filename}")

    def record(
        self,
        elapsed_time,
        north,
        east,
        down,
        battery,
        mode
    ):

        if self.closed:
            return

        self.writer.writerow([
            f"{elapsed_time:.2f}",
            f"{north:.2f}",
            f"{east:.2f}",
            f"{down:.2f}",
            f"{battery:.1f}",
            mode
        ])

        self.file.flush()

    def close(self):

        if self.closed:
            return

        self.closed = True

        self.file.close()

        print("[RECORDER] Flight log saved")

import csv
import glob
import math
import os


class MissionReport:

    def __init__(self, log_directory="Logs"):
        self.log_directory = log_directory

    def latest_flight_log(self):
        files = glob.glob(
            os.path.join(
                self.log_directory,
                "flight_data_*.csv"
            )
        )

        if not files:
            return None

        return max(files, key=os.path.getmtime)

    def load_data(self, filename):
        data = []

        with open(filename, "r", newline="") as file:
            reader = csv.DictReader(file)

            for row in reader:
                try:
                    data.append({
                        "time": float(row["Time"]),
                        "north": float(row["North"]),
                        "east": float(row["East"]),
                        "down": float(row["Down"]),
                        "battery": float(row["Battery(%)"]),
                        "mode": row["FlightMode"],
                    })

                except (ValueError, KeyError):
                    continue

        return data

    @staticmethod
    def calculate_distance(data):
        total_distance = 0.0

        for previous, current in zip(data, data[1:]):
            dn = current["north"] - previous["north"]
            de = current["east"] - previous["east"]
            dd = current["down"] - previous["down"]

            total_distance += math.sqrt(
                dn * dn +
                de * de +
                dd * dd
            )

        return total_distance

    def generate(self):
        filename = self.latest_flight_log()

        print("\n========================================")
        print("        ODYSSEY POST-FLIGHT REPORT")
        print("========================================")

        if filename is None:
            print("No flight recorder data found.")
            print("========================================")
            return

        data = self.load_data(filename)

        if not data:
            print("Flight log contains no valid data.")
            print("========================================")
            return

        first = data[0]
        last = data[-1]

        duration = max(
            0.0,
            last["time"] - first["time"]
        )

        max_altitude = max(
            0.0,
            max(-row["down"] for row in data)
        )

        min_battery = min(
            row["battery"] for row in data
        )

        start_battery = first["battery"]
        end_battery = last["battery"]

        battery_used = max(
            0.0,
            start_battery - end_battery
        )

        min_north = min(
            row["north"] for row in data
        )

        max_north = max(
            row["north"] for row in data
        )

        min_east = min(
            row["east"] for row in data
        )

        max_east = max(
            row["east"] for row in data
        )

        total_distance = self.calculate_distance(data)

        horizontal_from_origin = math.sqrt(
            last["north"] ** 2 +
            last["east"] ** 2
        )

        print(f"Flight Log          : {filename}")
        print(f"Samples Recorded    : {len(data)}")
        print(f"Recorded Duration   : {duration:.2f} s")
        print(f"Maximum Altitude    : {max_altitude:.2f} m")
        print(f"Recorded Distance   : {total_distance:.2f} m")

        print("----------------------------------------")

        print(
            f"North Range         : "
            f"{min_north:.2f} to {max_north:.2f} m"
        )

        print(
            f"East Range          : "
            f"{min_east:.2f} to {max_east:.2f} m"
        )

        print(
            f"Last Position       : "
            f"N={last['north']:.2f}, "
            f"E={last['east']:.2f}, "
            f"D={last['down']:.2f}"
        )

        print(
            f"Horizontal Return   : "
            f"{horizontal_from_origin:.2f} m from origin"
        )

        print("----------------------------------------")

        print(f"Start Battery       : {start_battery:.1f}%")
        print(f"End Battery         : {end_battery:.1f}%")
        print(f"Minimum Battery     : {min_battery:.1f}%")
        print(f"Battery Used        : {battery_used:.1f}%")

        print("----------------------------------------")

        print(f"Last Recorded State : {last['mode']}")
        print("Data Source         : PX4 SITL telemetry")
        print("Coordinate Frame    : Local NED (m)")
        print("Simulation Platform : PX4 SITL + Gazebo")

        print("========================================")


if __name__ == "__main__":
    report = MissionReport()
    report.generate()
